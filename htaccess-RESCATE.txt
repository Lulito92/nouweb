
# .htaccess de RESCATE — 2026-07-23
# Version minima y segura. Sin compresion, sin cache, sin bloqueos.
# Solo lo imprescindible para que la web funcione.

RewriteEngine On

# Forzar HTTPS SIN bucle: la segunda condicion es la que evita el cuelgue
# en hostings con proxy SSL (Hostinger). NO quitarla.
RewriteCond %{HTTPS} off
RewriteCond %{HTTP:X-Forwarded-Proto} !https
RewriteRule ^(.*)$ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]

RewriteRule ^es/?$ / [L,R=301]
RewriteRule ^es/index\.html$ / [L,R=301]

DirectoryIndex index.html dashboard.php

Options -Indexes
